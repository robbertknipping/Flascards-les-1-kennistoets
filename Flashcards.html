    <script>

        // De Data van de flashcards

        // Let op: Dubbele backslashes (\\) zijn nodig voor de formules in Javascript strings!

        const cards = [

            { front: "Statische rekoefeningen", back: "Rekoefeningen waarbij je stilstaat en een spier langzaam oprekt (vaak tijdens de cooling-down)." },

            { front: "Grondmotorische eigenschappen", back: "De vijf basiseigenschappen van fitheid: Coördinatie, Lenigheid, Uithoudingsvermogen, Kracht, Snelheid." },

            { front: "Eurofittest: Sit-and-reach test meet...", back: "Lenigheid (flexibiliteit van de onderrug en hamstrings)." },

        let currentIndex = 0;

        

        // Elementen ophalen

        const cardInner = document.getElementById('card-inner');

        const frontText = document.getElementById('front-text');

        const backText = document.getElementById('back-text');

        const progress = document.getElementById('progress');

        const prevBtn = document.getElementById('prevBtn');

        const nextBtn = document.getElementById('nextBtn');


        // Functie om de kaart content te updaten

        function updateCard() {

            // Content invullen

            frontText.innerHTML = cards[currentIndex].front;

            backText.innerHTML = cards[currentIndex].back;

            

            // Progress update

            progress.innerText = `Kaart ${currentIndex + 1} van ${cards.length}`;

            

            // Knoppen aan/uit zetten

            prevBtn.disabled = currentIndex === 0;

            nextBtn.disabled = currentIndex === cards.length - 1;


            // MathJax vertellen dat er nieuwe formules zijn om te renderen

            if (window.MathJax && window.MathJax.typesetPromise) {

                window.MathJax.typesetPromise([frontText, backText]).catch((err) => console.log('MathJax error:', err));

            }

        }


        // Functie om kaart om te draaien

        function flipCard() {

            cardInner.classList.toggle('flipped');

        }


        // Volgende kaart logic

        function nextCard() {

            // Stop propagation (voorkom dat hij flipt als je op de knop drukt)

            event.stopPropagation();


            if (currentIndex < cards.length - 1) {

                // Als de kaart gedraaid is (achterkant zichtbaar), draai hem eerst terug

                if (cardInner.classList.contains('flipped')) {

                    cardInner.classList.remove('flipped');

                    // Wacht heel even met nieuwe content laden tot de kaart 'dicht' is (300ms)

                    setTimeout(() => {

                        currentIndex++;

                        updateCard();

                    }, 300);

                } else {

                    // Als hij al aan de voorkant is, direct updaten

                    currentIndex++;

                    updateCard();

                }

            }

        }


        // Vorige kaart logic

        function prevCard() {

            event.stopPropagation();


            if (currentIndex > 0) {

                if (cardInner.classList.contains('flipped')) {

                    cardInner.classList.remove('flipped');

                    setTimeout(() => {

                        currentIndex--;

                        updateCard();

                    }, 300);

                } else {

                    currentIndex--;

                    updateCard();

                }

            }

        }


        // Toetsenbord ondersteuning (Pijltjes + Spatie)

        document.addEventListener('keydown', (e) => {

            if (e.key === 'ArrowRight') {

                if(currentIndex < cards.length - 1) {

                    if (cardInner.classList.contains('flipped')) {

                        cardInner.classList.remove('flipped');

                        setTimeout(() => { currentIndex++; updateCard(); }, 300);

                    } else {

                        currentIndex++; updateCard();

                    }

                }

            }

            if (e.key === 'ArrowLeft') {

                if(currentIndex > 0) {

                     if (cardInner.classList.contains('flipped')) {

                        cardInner.classList.remove('flipped');

                        setTimeout(() => { currentIndex--; updateCard(); }, 300);

                    } else {

                        currentIndex--; updateCard();

                    }

                }

            }

            if (e.key === ' ' || e.key === 'Enter') {

                e.preventDefault(); 

                flipCard();

            }

        });


        // Start de applicatie

        updateCard();


    </script>
